function esc(s){
  return String(s)
    .replace(/&/g,'&amp;')
    .replace(/</g,'&lt;')
    .replace(/>/g,'&gt;')
    .replace(/"/g,'&quot;')
    .replace(/'/g,'&#39;');
}
/* ИСПРАВЛЕНО: равномерный shuffle (Fisher–Yates) */
function shuffle(arr){
  const a = arr.slice();
  for(let i=a.length-1;i>0;i--){ const j=Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; }
  return a;
}
function pad2(n){ return String(n).padStart(2,'0'); }

}

/* ИСПРАВЛЕНО: таймер сохраняется и восстанавливается; используется deadline, а не тик */
function stopTimer(){
  if(timerInt){ clearInterval(timerInt); timerInt=null; }
  const b = document.getElementById('examTimer');
  if(b){ b.textContent=''; b.style.display='none'; }
}
function startExamTimer(){
  if(mode!=='exam' || timerInt) return;
  if(!examDeadline){
    examLeft = EXAM_SECONDS;
    examDeadline = Date.now() + EXAM_SECONDS*1000;
  }
  updateTimerBadge();
  timerInt = setInterval(()=>{
    examLeft = Math.max(0, Math.round((examDeadline - Date.now())/1000));
    updateTimerBadge();
    if(examLeft<=0){ stopTimer(); autoFinish(); }
  }, 500);
}
function updateTimerBadge(){
  const el = document.getElementById('examTimer');
  if(!el) return;
  if(mode!=='exam'){ el.style.display='none'; return; }
  el.style.display='inline-block';
  const m = Math.floor(Math.max(0,examLeft)/60);
  const s = Math.max(0,examLeft)%60;
  el.textContent = '⏱ '+m+':'+pad2(s);
  el.style.color = examLeft<60 ? 'var(--bad)' : '';
}

/* ================= СТАРТ/ВОССТАНОВЛЕНИЕ ================= */
function startGame(m){
  const topics = getSelectedTopics();
  const cnt = getSelectedCount();
  if(topics.length===0){ alert('Выберите хотя бы одну тему.'); return; }
  settings = { topics, count: cnt };
  mode = m;
  clearProgress();
  let pool = CASES.map((c,i)=>({c,i})).filter(x=>topics.includes(x.c.topic));
  pool = shuffle(pool).map(x=>x.i);
  if(cnt>0 && cnt<pool.length) pool = pool.slice(0, cnt);
  order = pool;
  cur=0; step=0; totalScore=0; clock=0; caseTime=0; revealed=new Set(); chosen=new Set(); hintUsed=0;
  examDeadline=0; examLeft=0; caseResults=[];
  render();
}
function resumeGame(){
  const d = loadProgress(); if(!d) return startGame('study');
  mode = d.mode||'study'; order = d.order; cur = d.cur; totalScore = d.totalScore;
  clock = d.clock||0; caseTime = d.caseTime||0;
  step = d.step||0;
  hintUsed = d.hintUsed||0;
  revealed = new Set(d.revealed||[]);
  chosen = new Set(d.chosen||[]);
  examLeft = d.examLeft||0;
  examDeadline = d.examDeadline||0;
  caseResults = d.caseResults||[];
  settings = d.settings || { topics: null, count: null };
  // если шаг был на результатах — покажем результат повторно
  render();
}

/* ИСПРАВЛЕНО: прогресс учитывает текущий шаг и доходит до 100% */
function updateProgBar(percent){
  const p = Math.max(0, Math.min(100, percent));
  const bar = document.getElementById('prog');
  if(bar) bar.style.width = p+'%';
  const pb = document.getElementById('progBar');
  if(pb) pb.setAttribute('aria-valuenow', String(Math.round(p)));
}

  document.getElementById('clock').textContent = clock;

  // прогресс: (cur + step/4)/order.length * 100

  else if(mode==='exam') stopTimer();
}

/* ================= ЗАВЕРШЕНИЕ КЕЙСА ================= */
function finishCase(c, txIdx, auto){
  let sc = 0; const parts=[];
  if(auto){ parts.push('⏱ Время вышло — кейс завершён автоматически; невыполненные этапы получили 0 баллов.'); }
  const ess = c.tests.map((t,i)=>t.ess?i:-1).filter(i=>i>=0);
  const essDone = ess.filter(i=>revealed.has(i)).length;
  const extra = [...revealed].filter(i=>!c.tests[i].ess && !c.tests[i].harmful).length;
  const harmful = [...revealed].filter(i=>c.tests[i].harmful).length;
  let testPts = essDone*4 - extra*2 - harmful*8;
  testPts = Math.min(20, Math.max(0, testPts));
  sc += testPts;
  parts.push(`<b>Обследования: ${testPts}/20</b> — информативных ключевых: ${essDone}/${ess.length}; лишних: ${extra}; опасных/бесполезных: ${harmful}.`);

  let dxPts=0;
  const main=[...chosen][0];
  if(main===c.dxMain){ dxPts+=40; parts.push(`<b>Диагноз: 40/40</b> — основной диагноз верный: «${esc(c.dxMain)}».`); }
  else if(chosen.has(c.dxMain)){ dxPts+=15; parts.push(`<b>Диагноз: 15/50</b> — верный диагноз был в альтернативах («${esc(c.dxMain)}»), но не основным.`); }
  else { parts.push(`<b>Диагноз: 0/50</b> — верный диагноз: «${esc(c.dxMain)}».`); }
  if(chosen.size>=3) dxPts+=10;
  dxPts = Math.min(50, dxPts);
  sc += dxPts;
  if(chosen.size>=3) parts.push(`Учтены альтернативы (+10) — хороший дифференциальный подход.`);

  const txOk = txIdx>=0 && c.tx[txIdx].ok;
  const txPts = txOk?30:0;
  sc += txPts;
  parts.push(txOk?`<b>Лечение: 30/30</b> — тактика верная.`:`<b>Лечение: 0/30</b> — выбранная тактика ошибочна/опасна.`);

  if(hintUsed && mode==='study'){ sc = Math.max(0, sc-2); parts.push('💡 Использована подсказка ординатуры (−2 балла).'); }

  caseResults.push({ idx: order[cur], title: c.hiddenTitle, score: sc, parts, time: caseTime, chosenMain: main||'—' });
  totalScore += sc;
  saveProgress(); stopTimer();
  step=4; renderResults(c, sc, parts);
}

    <div class="mut">«${esc(c.hiddenTitle)}»</div>

}

  const errors = caseResults.filter(r=>r.score<70);
  app.innerHTML = `

    <p><b>${esc(grade)}</b></p>
  </div>
  <div class="card"><h3>Рекомендации для роста</h3><p>${esc(advice)}</p>
    <p class="mut" style="margin-top:8px">Направления, в которых стоит усилить подготовку: острые цереброваскулярные заболевания, нейроинфекции, нейромышечные неотложки, эпилептология, демиелинизирующие заболевания, радикулопатии и позвоночная патология.</p></div>

}

function restart(){ clearProgress(); cur=0; step=0; totalScore=0; clock=0; revealed=new Set(); chosen=new Set(); hintUsed=0; caseResults=[]; examDeadline=0; examLeft=0; startScreen(); }

function repeatErrors(){
  const idxs = caseResults.filter(r=>r.score<70).map(r=>r.idx);
  if(idxs.length===0) return;
  order = shuffle(idxs);
  cur=0; step=0; totalScore=0; clock=0; caseTime=0; revealed=new Set(); chosen=new Set(); hintUsed=0;
  caseResults=[]; examDeadline=0; examLeft=0;
  mode='study';
  clearProgress(); render();
}

/* ================= ЭКСПОРТ ОТЧЁТА ================= */
function exportReport(){
  const lines = [];
  lines.push('НЕВРОЛОГ-СТАЖЁР — ОТЧЁТ О СМЕНЕ');
  lines.push('Дата: ' + fmtDate(new Date()));
  lines.push('Режим: ' + (mode==='exam'?'экзаменационный':'учебный'));
  lines.push('Кейсов: ' + caseResults.length);
  lines.push('Балл: ' + totalScore + ' / ' + (order.length*100));
  lines.push('Время: ' + clock + ' мин');
  lines.push('='.repeat(50));
  lines.push('');
  caseResults.forEach((r,i)=>{
    lines.push(`[${i+1}] ${r.title}`);
    lines.push(`    Балл: ${r.score}/100 · время: ${r.time} мин`);
    lines.push(`    Основной диагноз студента: ${r.chosenMain}`);
    lines.push('    Разбор:');
    r.parts.forEach(p=>{
      // уберём HTML-теги из частей
      const plain = p.replace(/<[^>]+>/g,'');
      lines.push('      - ' + plain);
    });
    lines.push('');
  });
  const blob = new Blob([lines.join('\n')], {type:'text/plain;charset=utf-8'});
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'nevrolog-report-' + new Date().toISOString().slice(0,10) + '.txt';
  document.body.appendChild(a);
  a.click();
  setTimeout(()=>{ URL.revokeObjectURL(url); a.remove(); }, 1000);
}

/* ================= PWA: manifest + service worker (inline) ================= */
/* Регистрируем SW через Blob, чтобы приложению не нужны были отдельные файлы. */
(function registerPWA(){
  if(!('serviceWorker' in navigator)) return;
  // SW с версией — обновляйте при изменениях
  const swCode = `
    const CACHE = 'neurosim-v3';
    self.addEventListener('install', e=>{ self.skipWaiting(); });
    self.addEventListener('activate', e=>{
      e.waitUntil(caches.keys().then(keys=>Promise.all(keys.filter(k=>k!==CACHE).map(k=>caches.delete(k)))).then(()=>self.clients.claim()));
    });
    self.addEventListener('fetch', e=>{
      const req = e.request;
      if(req.method!=='GET') return;
      // network-first для документа, cache-first для остального
      if(req.mode==='navigate'){
        e.respondWith(fetch(req).then(r=>{ caches.open(CACHE).then(c=>c.put(req, r.clone())); return r; }).catch(()=>caches.match(req).then(r=>r||caches.match('/'))));
      } else {
        e.respondWith(caches.match(req).then(r=>r||fetch(req).then(resp=>{ caches.open(CACHE).then(c=>c.put(req, resp.clone())); return resp; })));
      }
    });
  `;
  try{
    const blob = new Blob([swCode], {type:'application/javascript'});
    const url = URL.createObjectURL(blob);
    navigator.serviceWorker.register(url).catch(()=>{ /* SW из blob иногда не работает в некоторых браузерах — не критично */ });
  }catch(e){ /* silent */ }

  // Динамический manifest через Blob (позволяет установку на телефон в Chrome/Edge)
  try{
    const manifest = {
      name: 'Невролог-стажёр — симулятор врача',
      short_name: 'Невролог',
      start_url: '.',
      display: 'standalone',
      background_color: '#0f172a',
      theme_color: '#0f172a',
      lang: 'ru',
      icons: [] // без отдельных PNG; для полноценной установки можно добавить SVG-иконку
    };
    const mBlob = new Blob([JSON.stringify(manifest)], {type:'application/manifest+json'});
    const mUrl = URL.createObjectURL(mBlob);
    const link = document.createElement('link');
    link.rel = 'manifest';
    link.href = mUrl;
    document.head.appendChild(link);
  }catch(e){ /* silent */ }
})();
